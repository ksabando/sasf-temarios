## Ejercicio 2: Sistema de Reservas de Hotel ⭐⭐⭐

### Modelo Relacional

**Tablas:**
1. `HABITACIONES` (id_habitacion PK, numero, tipo, estado, descripcion)
2. `CLIENTES` (id_cliente PK, nombre, email, telefono, direccion)
3. `TEMPORADAS` (id_temporada PK, nombre, fecha_inicio, fecha_fin)
4. `PRECIOS_TEMPORADA` (id_precio PK, temporada_id FK, tipo_habitacion, precio_noche)
5. `RESERVAS` (id_reserva PK, cliente_id FK, fecha_reserva, estado)
6. `DETALLE_RESERVA` (id_detalle PK, reserva_id FK, habitacion_id FK, fecha_checkin, fecha_checkout, costo_total)
7. `PAGOS_RESERVA` (id_pago PK, reserva_id FK, monto, fecha_pago, tipo)

### PL/SQL

```sql
-- Función calcular costo de reserva
CREATE OR REPLACE FUNCTION calcular_costo_reserva (
    p_fecha_checkin     IN DATE,
    p_fecha_checkout    IN DATE,
    p_tipo_habitacion   IN VARCHAR2
) RETURN NUMBER IS
    v_noches            NUMBER;
    v_costo_total       NUMBER := 0;
    v_precio_noche      NUMBER;
    v_fecha_actual      DATE;
BEGIN
    v_noches := TRUNC(p_fecha_checkout) - TRUNC(p_fecha_checkin);
    IF v_noches <= 0 THEN
        RAISE_APPLICATION_ERROR(-20001, 'Check-out debe ser posterior a check-in');
    END IF;

    v_fecha_actual := TRUNC(p_fecha_checkin);

    -- Iterar cada noche y obtener precio según temporada vigente
    FOR i IN 0 .. v_noches - 1 LOOP
        BEGIN
            SELECT pt.precio_noche INTO v_precio_noche
            FROM PRECIOS_TEMPORADA pt
            JOIN TEMPORADAS t ON pt.temporada_id = t.id_temporada
            WHERE pt.tipo_habitacion = p_tipo_habitacion
              AND v_fecha_actual BETWEEN t.fecha_inicio AND t.fecha_fin;

            v_costo_total := v_costo_total + v_precio_noche;
        EXCEPTION
            WHEN NO_DATA_FOUND THEN
                -- Si no hay temporada definida, usar precio por defecto
                v_costo_total := v_costo_total + 50000;
        END;
        v_fecha_actual := v_fecha_actual + 1;
    END LOOP;

    -- Descuento por estadía larga (> 7 noches = 10%)
    IF v_noches > 7 THEN
        v_costo_total := v_costo_total * 0.90;
    END IF;

    RETURN ROUND(v_costo_total, 2);
END calcular_costo_reserva;
/

-- Procedimiento cancelar reserva
CREATE OR REPLACE PROCEDURE cancelar_reserva (
    p_reserva_id IN NUMBER
) IS
    v_checkin       DATE;
    v_costo_total   NUMBER;
    v_penalizacion  NUMBER := 0;
    v_horas_diff    NUMBER;
BEGIN
    SELECT MIN(dr.fecha_checkin), SUM(dr.costo_total)
    INTO v_checkin, v_costo_total
    FROM DETALLE_RESERVA dr
    WHERE dr.reserva_id = p_reserva_id;

    -- Calcular horas hasta check-in
    v_horas_diff := (v_checkin - SYSDATE) * 24;

    IF v_horas_diff < 48 THEN
        -- Cancelación con menos de 48h: penalización 50%
        v_penalizacion := v_costo_total * 0.50;
    END IF;

    -- Actualizar estado de reserva
    UPDATE RESERVAS SET estado = 'CANCELADA'
    WHERE id_reserva = p_reserva_id;

    -- Registrar penalización si aplica
    IF v_penalizacion > 0 THEN
        INSERT INTO PAGOS_RESERVA (reserva_id, monto, tipo)
        VALUES (p_reserva_id, v_penalizacion, 'PENALIZACION');

        -- Liberar habitaciones
        FOR r IN (SELECT habitacion_id FROM DETALLE_RESERVA WHERE reserva_id = p_reserva_id) LOOP
            UPDATE HABITACIONES SET estado = 'DISPONIBLE'
            WHERE id_habitacion = r.habitacion_id;
        END LOOP;
    END IF;

    COMMIT;
    DBMS_OUTPUT.PUT_LINE('Reserva cancelada. Penalización: $' || v_penalizacion);
END cancelar_reserva;
/

-- Trigger anti-overbooking
CREATE OR REPLACE TRIGGER trg_evitar_overbooking
BEFORE INSERT OR UPDATE ON DETALLE_RESERVA
FOR EACH ROW
DECLARE
    v_count NUMBER;
BEGIN
    SELECT COUNT(*) INTO v_count
    FROM DETALLE_RESERVA dr
    JOIN RESERVAS r ON dr.reserva_id = r.id_reserva
    WHERE dr.habitacion_id = :NEW.habitacion_id
      AND r.estado NOT IN ('CANCELADA', 'COMPLETADA')
      AND dr.id_detalle != NVL(:NEW.id_detalle, 0)
      AND (
          (:NEW.fecha_checkin BETWEEN dr.fecha_checkin AND dr.fecha_checkout - 1)
          OR
          (:NEW.fecha_checkout - 1 BETWEEN dr.fecha_checkin AND dr.fecha_checkout - 1)
          OR
          (dr.fecha_checkin BETWEEN :NEW.fecha_checkin AND :NEW.fecha_checkout - 1)
      );

    IF v_count > 0 THEN
        RAISE_APPLICATION_ERROR(-20002,
            'Overbooking: Habitación no disponible en las fechas solicitadas.');
    END IF;
END;
/
```

```sql
-- Función calcular líquido para un empleado en un mes
CREATE OR REPLACE FUNCTION calcular_liquido (
    p_empleado_id   IN NUMBER,
    p_mes           IN NUMBER,
    p_anio          IN NUMBER
) RETURN NUMBER IS
    v_emp           EMPLEADOS%ROWTYPE;
    v_bono_antig    NUMBER := 0;
    v_total_he      NUMBER := 0;
    v_total_haberes NUMBER;
    v_desc_salud    NUMBER;
    v_desc_pension  NUMBER;
    v_desc_imp      NUMBER := 0;
    v_total_desc    NUMBER;
    v_liquido       NUMBER;
    v_valor_hora    NUMBER;
    v_porc_salud    NUMBER;
    v_porc_pension  NUMBER;
    v_porc_imp      NUMBER;
    v_bono_anual    NUMBER;
    v_tope_bono     NUMBER;
    v_factor_he_h   NUMBER;
    v_factor_he_fs  NUMBER;
    v_anios_antig   NUMBER;
BEGIN
    -- Obtener datos del empleado
    SELECT * INTO v_emp FROM EMPLEADOS WHERE id_empleado = p_empleado_id;

    -- Obtener parámetros
    SELECT valor INTO v_porc_salud   FROM PARAMETROS_NOMINA WHERE nombre = 'PORC_SALUD';
    SELECT valor INTO v_porc_pension FROM PARAMETROS_NOMINA WHERE nombre = 'PORC_PENSION';
    SELECT valor INTO v_porc_imp     FROM PARAMETROS_NOMINA WHERE nombre = 'PORC_IMP_HONORARIOS';
    SELECT valor INTO v_bono_anual   FROM PARAMETROS_NOMINA WHERE nombre = 'BONO_ANTIGUEDAD_ANUAL';
    SELECT valor INTO v_tope_bono    FROM PARAMETROS_NOMINA WHERE nombre = 'TOPE_BONO_ANTIGUEDAD';
    SELECT valor INTO v_factor_he_h  FROM PARAMETROS_NOMINA WHERE nombre = 'FACTOR_HE_HABIL';
    SELECT valor INTO v_factor_he_fs FROM PARAMETROS_NOMINA WHERE nombre = 'FACTOR_HE_FIN_SEMANA';

    -- Valor hora base (asumiendo 180 horas mensuales)
    v_valor_hora := v_emp.sueldo_base / 180;

    -- Calcular bono antigüedad (solo planta)
    IF v_emp.tipo_contrato = 'PLANTA' THEN
        v_anios_antig := TRUNC(MONTHS_BETWEEN(
            TO_DATE(p_anio || '-' || LPAD(p_mes,2,'0') || '-01', 'YYYY-MM-DD'),
            v_emp.fecha_ingreso) / 12);
        v_bono_antig := LEAST(v_anios_antig * v_bono_anual, v_tope_bono) * v_emp.sueldo_base;
    END IF;

    -- Calcular horas extra del mes
    SELECT NVL(SUM(
        CASE tipo_dia
            WHEN 'HABIL'      THEN horas * v_valor_hora * v_factor_he_h
            WHEN 'FIN_SEMANA' THEN horas * v_valor_hora * v_factor_he_fs
        END), 0)
    INTO v_total_he
    FROM HORAS_EXTRA
    WHERE empleado_id = p_empleado_id
      AND EXTRACT(MONTH FROM fecha) = p_mes
      AND EXTRACT(YEAR FROM fecha) = p_anio;

    -- Total haberes
    v_total_haberes := v_emp.sueldo_base + v_bono_antig + v_total_he;

    -- Descuentos
    v_desc_salud := v_total_haberes * v_porc_salud;

    IF v_emp.tipo_contrato = 'HONORARIOS' THEN
        v_desc_pension := 0; -- Honorarios no tienen descuento pensión
        v_desc_imp := v_total_haberes * v_porc_imp;
    ELSE
        v_desc_pension := v_total_haberes * v_porc_pension;
    END IF;

    v_total_desc := v_desc_salud + v_desc_pension + v_desc_imp;
    v_liquido := v_total_haberes - v_total_desc;

    RETURN ROUND(v_liquido, 0);
END calcular_liquido;
/

-- Procedimiento generar nómina mensual
CREATE OR REPLACE PROCEDURE generar_nomina_mensual (
    p_mes   IN NUMBER,
    p_anio  IN NUMBER
) IS
    v_liquido NUMBER;
BEGIN
    FOR emp IN (SELECT * FROM EMPLEADOS WHERE activo = 'S') LOOP
        v_liquido := calcular_liquido(emp.id_empleado, p_mes, p_anio);

        MERGE INTO NOMINA n
        USING (SELECT emp.id_empleado AS emp_id, p_mes AS m, p_anio AS a FROM DUAL) src
        ON (n.empleado_id = src.emp_id AND n.mes = src.m AND n.anio = src.a)
        WHEN MATCHED THEN UPDATE SET sueldo_liquido = v_liquido
        WHEN NOT MATCHED THEN INSERT (empleado_id, mes, anio, sueldo_liquido)
            VALUES (emp.id_empleado, p_mes, p_anio, v_liquido);
    END LOOP;
    COMMIT;

    DBMS_OUTPUT.PUT_LINE('Nómina generada para ' || p_mes || '/' || p_anio);
END generar_nomina_mensual;
/
```

```sql
        WHERE producto_id = r.producto_id

          AND estado = 'PENDIENTE';
```

```sql
-- Procedimiento cambiar de plan
CREATE OR REPLACE PROCEDURE cambiar_plan (
    p_cliente_id    IN NUMBER,
    p_nuevo_plan_id IN NUMBER
) IS
    v_plan_actual_id    NUMBER;
    v_precio_actual     NUMBER;
    v_precio_nuevo      NUMBER;
    v_dias_restantes    NUMBER;
    v_cargo_proporcional NUMBER := 0;
BEGIN
    SELECT cs.plan_id, p.precio_mensual
    INTO v_plan_actual_id, v_precio_actual
    FROM CLIENTES_SaaS cs
    JOIN PLANES p ON cs.plan_id = p.id_plan
    WHERE cs.id_cliente = p_cliente_id;

    SELECT precio_mensual INTO v_precio_nuevo FROM PLANES WHERE id_plan = p_nuevo_plan_id;

    -- Calcular días restantes del ciclo (asumiendo ciclos de 30 días)
    SELECT NVL(MAX(30 - (SYSDATE - fecha_factura)), 30) INTO v_dias_restantes
    FROM FACTURACION_SaaS
    WHERE cliente_id = p_cliente_id
      AND estado_pago = 'PAGADO'
      AND fecha_factura >= ADD_MONTHS(SYSDATE, -1);

    -- UPGRADE: cobro proporcional por la diferencia
    IF v_precio_nuevo > v_precio_actual THEN
        v_cargo_proporcional := (v_precio_nuevo - v_precio_actual) * (v_dias_restantes / 30);
        INSERT INTO FACTURACION_SaaS (cliente_id, plan_id, monto, estado_pago)
        VALUES (p_cliente_id, p_nuevo_plan_id, ROUND(v_cargo_proporcional, 0), 'PENDIENTE');
    END IF;

    -- Actualizar plan del cliente (downgrade aplica al siguiente ciclo)
    UPDATE CLIENTES_SaaS SET plan_id = p_nuevo_plan_id WHERE id_cliente = p_cliente_id;

    COMMIT;
    DBMS_OUTPUT.PUT_LINE('Plan cambiado. Cargo proporcional: $' || ROUND(v_cargo_proporcional, 0));
END cambiar_plan;
/

-- Función clientes activos
CREATE OR REPLACE FUNCTION clientes_activos RETURN SYS_REFCURSOR IS
    v_cursor SYS_REFCURSOR;
BEGIN
```

```sql
-- Trigger: validar transición de estados
CREATE OR REPLACE TRIGGER trg_validar_transicion_estado
BEFORE UPDATE OF estado ON PEDIDOS
FOR EACH ROW
DECLARE
    v_orden_ok BOOLEAN := FALSE;
BEGIN
    -- Definir transiciones válidas
    IF :OLD.estado = 'PENDIENTE' AND :NEW.estado IN ('CONFIRMADO','CANCELADO') THEN
        v_orden_ok := TRUE;
    ELSIF :OLD.estado = 'CONFIRMADO' AND :NEW.estado = 'EN_PREPARACION' THEN
        v_orden_ok := TRUE;
    ELSIF :OLD.estado = 'EN_PREPARACION' AND :NEW.estado = 'EN_CAMINO' THEN
        v_orden_ok := TRUE;
    ELSIF :OLD.estado = 'EN_CAMINO' AND :NEW.estado = 'ENTREGADO' THEN
        v_orden_ok := TRUE;
    END IF;

    IF NOT v_orden_ok THEN
        RAISE_APPLICATION_ERROR(-20003,
            'Transición inválida: de ' || :OLD.estado || ' a ' || :NEW.estado);
    END IF;

    -- Registrar historial
    INSERT INTO HISTORIAL_ESTADOS (pedido_id, estado_anterior, estado_nuevo)
    VALUES (:NEW.id_pedido, :OLD.estado, :NEW.estado);
END;

-- Función calcular ganancia del repartidor
CREATE OR REPLACE FUNCTION calcular_ganancia_repartidor (
    p_repartidor_id IN NUMBER,
    p_fecha         IN DATE DEFAULT SYSDATE
) RETURN NUMBER IS
    v_total NUMBER := 0;
    v_bono  NUMBER := 0;
BEGIN
    SELECT NVL(SUM(p.tarifa_envio), 0)
    INTO v_total
    FROM PEDIDOS p
    WHERE p.repartidor_id = p_repartidor_id
      AND TRUNC(p.fecha_pedido) = TRUNC(p_fecha)
      AND p.estado = 'ENTREGADO';

    -- Bono por pedidos con distancia > 5km (clientes en zona diferente al repartidor)
    SELECT NVL(SUM(
        CASE WHEN c.zona_id != r.zona_id THEN 2000 ELSE 0 END
    ), 0)
    INTO v_bono
    FROM PEDIDOS p
    JOIN CLIENTES_DELIVERY c ON p.cliente_id = c.id_cliente
    JOIN REPARTIDORES r ON p.repartidor_id = r.id_repartidor
    WHERE p.repartidor_id = p_repartidor_id
      AND TRUNC(p.fecha_pedido) = TRUNC(p_fecha)
      AND p.estado = 'ENTREGADO';

    RETURN v_total + v_bono;
END calcular_ganancia_repartidor;
/

-- Procedimiento asignar repartidor más cercano disponible
CREATE OR REPLACE PROCEDURE asignar_repartidor (
    p_pedido_id IN NUMBER
) IS
    v_zona_pedido   NUMBER;
    v_repartidor_id NUMBER;
BEGIN
    SELECT z.id_zona INTO v_zona_pedido
    FROM PEDIDOS p
    JOIN CLIENTES_DELIVERY c ON p.cliente_id = c.id_cliente
    JOIN ZONAS z ON c.zona_id = z.id_zona
    WHERE p.id_pedido = p_pedido_id;

    -- Buscar repartidor disponible en la misma zona
    BEGIN
        SELECT id_repartidor INTO v_repartidor_id
        FROM REPARTIDORES
        WHERE zona_id = v_zona_pedido AND disponible = 'S'
        AND ROWNUM = 1;
    EXCEPTION
        WHEN NO_DATA_FOUND THEN
            -- Si no hay en la misma zona, buscar cualquiera disponible
            SELECT id_repartidor INTO v_repartidor_id
            FROM REPARTIDORES
            WHERE disponible = 'S'
            AND ROWNUM = 1;
    END;

    UPDATE PEDIDOS SET repartidor_id = v_repartidor_id
    WHERE id_pedido = p_pedido_id;

    UPDATE REPARTIDORES SET disponible = 'N'
    WHERE id_repartidor = v_repartidor_id;

    COMMIT;
    DBMS_OUTPUT.PUT_LINE('Repartidor ' || v_repartidor_id || ' asignado al pedido ' || p_pedido_id);
END asignar_repartidor;
/
```

```sql
-- Función calcular nota final
CREATE OR REPLACE FUNCTION calcular_nota_final (
    p_curso_id      IN NUMBER,
    p_estudiante_id IN NUMBER
) RETURN NUMBER IS
    v_nota_final    NUMBER := 0;
    v_inscripcion_id NUMBER;
BEGIN
    SELECT id_inscripcion INTO v_inscripcion_id
    FROM INSCRIPCIONES
    WHERE curso_id = p_curso_id
      AND estudiante_id = p_estudiante_id
      AND estado IN ('CURSANDO', 'RECUPERACION');

    SELECT NVL(SUM(n.nota * e.porcentaje / 100), 0)
    INTO v_nota_final
    FROM NOTAS n
    JOIN EVALUACIONES e ON n.evaluacion_id = e.id_evaluacion
    WHERE n.inscripcion_id = v_inscripcion_id;

    RETURN ROUND(v_nota_final, 2);
END calcular_nota_final;
```

```sql
-- Procedimiento asignar ticket con round-robin
CREATE OR REPLACE PROCEDURE asignar_ticket (
    p_ticket_id IN NUMBER
) IS
    v_categoria_id  NUMBER;
    v_agente_id     NUMBER;
BEGIN
    SELECT categoria_id INTO v_categoria_id FROM TICKETS WHERE id_ticket = p_ticket_id;

    -- Round-robin: agente de la categoría con el ticket asignado más antiguo
    BEGIN
        SELECT id_agente INTO v_agente_id
        FROM (
            SELECT id_agente
            FROM AGENTES
            WHERE categoria_id = v_categoria_id
              AND activo = 'S'
              AND es_supervisor = 'N'
            ORDER BY ultimo_ticket ASC NULLS FIRST
        ) WHERE ROWNUM = 1;
    EXCEPTION
        WHEN NO_DATA_FOUND THEN
            RAISE_APPLICATION_ERROR(-20004, 'No hay agentes disponibles para esta categoría');
    END;

    UPDATE TICKETS SET agente_id = v_agente_id, estado = 'EN_PROCESO'
    WHERE id_ticket = p_ticket_id;

    UPDATE AGENTES SET ultimo_ticket = SYSDATE WHERE id_agente = v_agente_id;
    COMMIT;
END asignar_ticket;
/

-- Función tickets vencidos SLA
CREATE OR REPLACE FUNCTION tickets_vencidos_sla RETURN SYS_REFCURSOR IS
    v_cursor SYS_REFCURSOR;
BEGIN
    OPEN v_cursor FOR
        SELECT t.id_ticket, t.asunto, t.prioridad, t.estado,
               t.fecha_creacion,
               ROUND((SYSDATE - t.fecha_creacion) * 24, 1) AS horas_transcurridas,
               sla.horas_max
        FROM TICKETS t
        JOIN SLA_CONFIG sla ON t.prioridad = sla.prioridad
        WHERE t.estado NOT IN ('CERRADO', 'RESUELTO')
          AND (SYSDATE - t.fecha_creacion) * 24 > sla.horas_max
        ORDER BY (SYSDATE - t.fecha_creacion) * 24 DESC;

    RETURN v_cursor;
END tickets_vencidos_sla;
/

-- Trigger: escalar si viola SLA
CREATE OR REPLACE TRIGGER trg_escalar_sla
AFTER UPDATE ON TICKETS
FOR EACH ROW
DECLARE
    v_horas_max NUMBER;
    v_horas_transc NUMBER;
BEGIN
    SELECT horas_max INTO v_horas_max FROM SLA_CONFIG
    WHERE prioridad = :NEW.prioridad;

    v_horas_transc := (SYSDATE - :NEW.fecha_creacion) * 24;

    IF v_horas_transc > v_horas_max AND :NEW.escalado = 'N' THEN
        UPDATE TICKETS SET escalado = 'S', estado = 'ESCALADO'
        WHERE id_ticket = :NEW.id_ticket;
        DBMS_OUTPUT.PUT_LINE('Ticket ' || :NEW.id_ticket || ' escalado por violación SLA');
    END IF;
END;
```

```sql
-- Función calcular copago
CREATE OR REPLACE FUNCTION calcular_copago (
    p_atencion_id IN NUMBER
) RETURN NUMBER IS
    v_costo_base        NUMBER;
    v_porc_cobertura    NUMBER := 0;
    v_monto_seguro      NUMBER := 0;
    v_copago            NUMBER;
    v_paciente_id       NUMBER;
    v_seguro_id         NUMBER;
    v_tipo_servicio     VARCHAR2(30);
BEGIN
    SELECT a.paciente_id, s.costo_base, s.tipo
    INTO v_paciente_id, v_costo_base, v_tipo_servicio
    FROM ATENCIONES a
    JOIN SERVICIOS s ON a.servicio_id = s.id_servicio
    WHERE a.id_atencion = p_atencion_id;

    -- Buscar seguro del paciente
    BEGIN
        SELECT seguro_id INTO v_seguro_id
        FROM PACIENTES WHERE id_paciente = v_paciente_id;
    EXCEPTION
        WHEN NO_DATA_FOUND THEN v_seguro_id := NULL;
    END;

    -- Si tiene seguro, buscar cobertura
    IF v_seguro_id IS NOT NULL THEN
        BEGIN
            SELECT porcentaje INTO v_porc_cobertura
            FROM COBERTURAS
            WHERE seguro_id = v_seguro_id
              AND tipo_servicio = v_tipo_servicio;
        EXCEPTION
            WHEN NO_DATA_FOUND THEN v_porc_cobertura := 0;
        END;
    END IF;

    v_monto_seguro := v_costo_base * (v_porc_cobertura / 100);
    v_copago := v_costo_base - v_monto_seguro;

    RETURN ROUND(v_copago, 0);
END calcular_copago;
/

-- Procedimiento generar factura
CREATE OR REPLACE PROCEDURE generar_factura (
    p_atencion_id IN NUMBER
) IS
    v_costo_base    NUMBER;
    v_copago        NUMBER;
    v_monto_seguro  NUMBER;
BEGIN
    v_copago := calcular_copago(p_atencion_id);

    SELECT s.costo_base INTO v_costo_base
    FROM ATENCIONES a
    JOIN SERVICIOS s ON a.servicio_id = s.id_servicio
    WHERE a.id_atencion = p_atencion_id;

    v_monto_seguro := v_costo_base - v_copago;

    INSERT INTO FACTURAS (atencion_id, monto_total, monto_seguro, copago_paciente)
    VALUES (p_atencion_id, v_costo_base, v_monto_seguro, v_copago);

    COMMIT;
    DBMS_OUTPUT.PUT_LINE('Factura generada. Total: $' || v_costo_base ||
        ' | Seguro: $' || v_monto_seguro || ' | Copago: $' || v_copago);
END generar_factura;
/
```

```sql
-- Trigger: extender subasta si puja en últimos 5 minutos
CREATE OR REPLACE TRIGGER trg_extender_subasta
BEFORE INSERT ON PUJAS
FOR EACH ROW
DECLARE
    v_fecha_fin     DATE;
    v_minutos_rest  NUMBER;
    v_precio_base   NUMBER;
    v_incremento    NUMBER;
    v_puja_max      NUMBER;
BEGIN
    SELECT fecha_fin, precio_base, NVL(incremento_min, precio_base * 0.05)
    INTO v_fecha_fin, v_precio_base, v_incremento
    FROM ITEMS_SUBASTA WHERE id_item = :NEW.item_id;

    -- Validar que la puja sea mayor al incremento mínimo
    SELECT NVL(MAX(monto), v_precio_base) INTO v_puja_max
    FROM PUJAS WHERE item_id = :NEW.item_id;

    IF :NEW.monto < v_puja_max + v_incremento THEN
        RAISE_APPLICATION_ERROR(-20006,
            'La puja debe ser al menos $' || (v_puja_max + v_incremento) ||
            ' (puja actual + incremento mínimo de $' || v_incremento || ')');
    END IF;

    -- Si la puja ocurre en los últimos 5 minutos, extender
    v_minutos_rest := (v_fecha_fin - SYSDATE) * 24 * 60;

    IF v_minutos_rest <= 5 AND v_minutos_rest > 0 THEN
        UPDATE ITEMS_SUBASTA
        SET fecha_fin = fecha_fin + INTERVAL '5' MINUTE,
            estado = 'EXTENDIDA'
        WHERE id_item = :NEW.item_id;
        DBMS_OUTPUT.PUT_LINE('Subasta extendida 5 minutos por puja en tiempo extra.');
    END IF;
END;
/

-- Procedimiento finalizar subasta
CREATE OR REPLACE PROCEDURE finalizar_subasta (
    p_item_id IN NUMBER
) IS
    v_ganador_id    NUMBER;
    v_monto_ganador NUMBER;
    v_puja_max      NUMBER;
BEGIN
    -- Determinar ganador (mayor puja)
    SELECT comprador_id, monto INTO v_ganador_id, v_puja_max
    FROM (
        SELECT comprador_id, monto
        FROM PUJAS
        WHERE item_id = p_item_id
        ORDER BY monto DESC
    ) WHERE ROWNUM = 1;

    -- Actualizar item con ganador
    UPDATE ITEMS_SUBASTA
    SET estado = 'FINALIZADA', ganador_id = v_ganador_id
    WHERE id_item = p_item_id;

    -- Registrar pago pendiente
    INSERT INTO PAGOS_SUBASTA (item_id, comprador_id, monto)
    VALUES (p_item_id, v_ganador_id, v_puja_max);

    COMMIT;
    DBMS_OUTPUT.PUT_LINE('Subasta finalizada. Ganador ID: ' || v_ganador_id ||
        ' | Monto: $' || v_puja_max);
END finalizar_subasta;

-- Función estado pago ganador (maneja no-pago y reasigna)
CREATE OR REPLACE FUNCTION estado_pago_ganador (
    p_item_id IN NUMBER
) RETURN VARCHAR2 IS
    v_pago_estado   VARCHAR2(20);
    v_estado        VARCHAR2(20);
    v_segundo_id    NUMBER;
    v_segundo_monto NUMBER;
BEGIN
    SELECT estado INTO v_pago_estado
    FROM PAGOS_SUBASTA WHERE item_id = p_item_id
    AND ROWNUM = 1
    ORDER BY id_pago DESC;

    -- Si no pagó en 48 horas, reasignar al segundo
    IF v_pago_estado = 'PENDIENTE' THEN
        SELECT (SYSDATE - fecha_pago) INTO v_estado
        FROM PAGOS_SUBASTA WHERE item_id = p_item_id
        AND ROWNUM = 1
        ORDER BY id_pago DESC;

        -- (simplificado: aquí se verificaría 48h reales)

        -- Reasignar al segundo mejor postor
        BEGIN
            SELECT comprador_id, monto INTO v_segundo_id, v_segundo_monto
            FROM (
                SELECT comprador_id, monto,
                       ROW_NUMBER() OVER (ORDER BY monto DESC) AS rn
                FROM PUJAS WHERE item_id = p_item_id
            ) WHERE rn = 2;

            UPDATE ITEMS_SUBASTA SET ganador_id = v_segundo_id
            WHERE id_item = p_item_id;

            INSERT INTO PAGOS_SUBASTA (item_id, comprador_id, monto)
            VALUES (p_item_id, v_segundo_id, v_segundo_monto);

            DBMS_OUTPUT.PUT_LINE('Ganador anterior no pagó. Reasignado al segundo.');
        EXCEPTION
            WHEN NO_DATA_FOUND THEN
                UPDATE ITEMS_SUBASTA SET estado = 'CANCELADA'
                WHERE id_item = p_item_id;
        END;
    END IF;

    RETURN v_pago_estado;
END estado_pago_ganador;
```
