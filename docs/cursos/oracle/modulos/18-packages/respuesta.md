## Respuesta 2: Package `pkg_reportes`

```sql
CREATE OR REPLACE PACKAGE pkg_reportes AS
    PROCEDURE generar_reporte(p_proyecto_id NUMBER);
    PROCEDURE generar_reporte;
END pkg_reportes;
/

CREATE OR REPLACE PACKAGE BODY pkg_reportes AS

    PROCEDURE generar_reporte(p_proyecto_id NUMBER) IS
        v_nombre      proyectos.nombre%TYPE;
        v_estado      proyectos.estado%TYPE;
        v_presupuesto proyectos.presupuesto%TYPE;
        v_total_costo NUMBER;
    BEGIN
        SELECT nombre, estado, presupuesto
        INTO v_nombre, v_estado, v_presupuesto
        FROM proyectos WHERE id = p_proyecto_id;

        DBMS_OUTPUT.PUT_LINE('========================================');
        DBMS_OUTPUT.PUT_LINE('REPORTE DE PROYECTO');
        DBMS_OUTPUT.PUT_LINE('========================================');
        DBMS_OUTPUT.PUT_LINE('Proyecto: ' || v_nombre);
        DBMS_OUTPUT.PUT_LINE('Estado:   ' || pkg_utilidades.estado_texto(v_estado));
        DBMS_OUTPUT.PUT_LINE('Presupuesto: ' || pkg_utilidades.formatear_moneda(NVL(v_presupuesto, 0)));
        DBMS_OUTPUT.PUT_LINE('----------------------------------------');
        DBMS_OUTPUT.PUT_LINE('TAREAS:');
        DBMS_OUTPUT.PUT_LINE(RPAD('ID', 6) || RPAD('Nombre', 30) || RPAD('Responsable', 20) ||
                             RPAD('Horas', 8) || RPAD('Costo', 12));
        DBMS_OUTPUT.PUT_LINE(RPAD('-', 6) || RPAD('-', 30) || RPAD('-', 20) ||
                             RPAD('-', 8) || RPAD('-', 12));

        FOR r IN (SELECT t.id, t.nombre, e.nombre AS responsable,
                         t.horas_reales, calcular_costo_real_tarea(t.id) AS costo
                  FROM tareas t
                  LEFT JOIN empleados e ON t.responsable_id = e.id
                  WHERE t.proyecto_id = p_proyecto_id
                  ORDER BY t.id)
        LOOP
            DBMS_OUTPUT.PUT_LINE(
                RPAD(r.id, 6) ||
                RPAD(SUBSTR(r.nombre, 1, 28), 30) ||
                RPAD(NVL(r.responsable, '(sin asignar)'), 20) ||
                RPAD(NVL(r.horas_reales, 0), 8) ||
                RPAD(pkg_utilidades.formatear_moneda(NVL(r.costo, 0), 'USD'), 12)
            );
            v_total_costo := NVL(v_total_costo, 0) + NVL(r.costo, 0);
        END LOOP;

        DBMS_OUTPUT.PUT_LINE('----------------------------------------');
        DBMS_OUTPUT.PUT_LINE('COSTO TOTAL: ' || pkg_utilidades.formatear_moneda(v_total_costo));
        DBMS_OUTPUT.PUT_LINE('========================================');
    END generar_reporte;

    PROCEDURE generar_reporte IS
    BEGIN
        DBMS_OUTPUT.PUT_LINE('========================================');
        DBMS_OUTPUT.PUT_LINE('REPORTE GLOBAL DE PROYECTOS');
        DBMS_OUTPUT.PUT_LINE('========================================');
        FOR r IN (SELECT id, nombre, estado FROM proyectos ORDER BY id) LOOP
            DBMS_OUTPUT.PUT_LINE('');
            generar_reporte(r.id);
        END LOOP;
    END generar_reporte;

END pkg_reportes;
/
```

## Respuesta 3: Package `pkg_utilidades`

### Tabla de log

```sql
CREATE TABLE log_errores (
    id       NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    mensaje  VARCHAR2(1000) NOT NULL,
    modulo   VARCHAR2(100) DEFAULT 'APLICACION',
    usuario  VARCHAR2(100),
    fecha    TIMESTAMP DEFAULT SYSTIMESTAMP
);

COMMENT ON TABLE log_errores IS 'Registro de errores de la aplicación';
```

### Package

```sql
CREATE OR REPLACE PACKAGE pkg_utilidades AS
    FUNCTION formatear_moneda(p_monto NUMBER, p_moneda VARCHAR2 DEFAULT 'USD') RETURN VARCHAR2;
    FUNCTION estado_texto(p_estado CHAR) RETURN VARCHAR2;
    PROCEDURE log_error(p_mensaje VARCHAR2, p_modulo VARCHAR2 DEFAULT 'APLICACION');
END pkg_utilidades;
/

CREATE OR REPLACE PACKAGE BODY pkg_utilidades AS

    FUNCTION formatear_moneda(p_monto NUMBER, p_moneda VARCHAR2 DEFAULT 'USD') RETURN VARCHAR2 IS
    BEGIN
        RETURN TO_CHAR(p_monto, 'FM$999,999,990.00') || ' ' || p_moneda;
    END formatear_moneda;

    FUNCTION estado_texto(p_estado CHAR) RETURN VARCHAR2 IS
    BEGIN
        RETURN CASE p_estado
                    WHEN 'P' THEN 'Pendiente'
                    WHEN 'E' THEN 'En Progreso'
                    WHEN 'C' THEN 'Completado'
                    ELSE 'Desconocido'
               END;
    END estado_texto;

    PROCEDURE log_error(p_mensaje VARCHAR2, p_modulo VARCHAR2 DEFAULT 'APLICACION') IS
        PRAGMA AUTONOMOUS_TRANSACTION;
    BEGIN
        INSERT INTO log_errores (mensaje, modulo, usuario)
        VALUES (p_mensaje, p_modulo, USER);
        COMMIT;
    END log_error;

END pkg_utilidades;
/
```

## Respuesta 4: Contador de llamadas

```sql
CREATE OR REPLACE PACKAGE BODY pkg_tareas AS
    -- Variable privada de sesión
    g_contador_llamadas NUMBER := 0;

    CURSOR c_tareas_activas RETURN tareas%ROWTYPE IS
        SELECT * FROM tareas WHERE estado IN ('P', 'E')
        ORDER BY fecha_inicio NULLS LAST;

    PROCEDURE asignar(p_tarea_id IN tareas.id%TYPE, p_empleado_id IN empleados.id%TYPE) IS
        -- ... (misma implementación del Ejercicio 1)
        v_empleado_existe  NUMBER;
        v_tarea_existe     NUMBER;
        v_estado_tarea     tareas.estado%TYPE;
        v_tareas_activas   NUMBER;
    BEGIN
        SELECT COUNT(*) INTO v_empleado_existe FROM empleados WHERE id = p_empleado_id;
        IF v_empleado_existe = 0 THEN
            RAISE_APPLICATION_ERROR(-20010, 'El empleado con ID ' || p_empleado_id || ' no existe');
        END IF;

        SELECT COUNT(*), estado INTO v_tarea_existe, v_estado_tarea
        FROM tareas WHERE id = p_tarea_id
        GROUP BY estado;

        IF v_tarea_existe = 0 THEN
            RAISE_APPLICATION_ERROR(-20011, 'La tarea con ID ' || p_tarea_id || ' no existe');
        END IF;

        IF v_estado_tarea != 'P' THEN
            RAISE_APPLICATION_ERROR(-20012, 'La tarea debe estar en estado Pendiente. Estado actual: ' || v_estado_tarea);
        END IF;

        SELECT COUNT(*) INTO v_tareas_activas
        FROM tareas
        WHERE responsable_id = p_empleado_id AND estado IN ('P', 'E');

        IF v_tareas_activas >= 5 THEN
            RAISE_APPLICATION_ERROR(-20013, 'El empleado ya tiene ' || v_tareas_activas || ' tareas activas');
        END IF;

        UPDATE tareas SET responsable_id = p_empleado_id, estado = 'E' WHERE id = p_tarea_id;
        g_contador_llamadas := g_contador_llamadas + 1;
        DBMS_OUTPUT.PUT_LINE('Tarea ' || p_tarea_id || ' asignada a empleado ' || p_empleado_id);
    END asignar;

    FUNCTION calcular_costo(p_tarea_id IN tareas.id%TYPE) RETURN NUMBER IS
    BEGIN
        RETURN calcular_costo_real_tarea(p_tarea_id);
    END calcular_costo;

    PROCEDURE listar_por_empleado(p_empleado_id IN empleados.id%TYPE) IS
    BEGIN
        DBMS_OUTPUT.PUT_LINE('=== Tareas del empleado ID ' || p_empleado_id || ' ===');
        FOR r IN (SELECT t.id, t.nombre, t.estado, t.horas_estimadas, t.horas_reales,
                         p.nombre AS proyecto
                  FROM tareas t
                  JOIN proyectos p ON t.proyecto_id = p.id
                  WHERE t.responsable_id = p_empleado_id
                  ORDER BY t.estado, t.fecha_inicio)
        LOOP
            DBMS_OUTPUT.PUT_LINE('T#' || r.id || ' | ' || r.nombre ||
                                 ' | ' || r.estado || ' | Proy: ' || r.proyecto ||
                                 ' | Est: ' || r.horas_estimadas || 'h | Real: ' || NVL(r.horas_reales, 0) || 'h');
        END LOOP;
    END listar_por_empleado;

    PROCEDURE mostrar_estadisticas IS
        v_total_tareas_activas NUMBER;
        v_total_empleados      NUMBER;
    BEGIN
        SELECT COUNT(*) INTO v_total_tareas_activas FROM tareas WHERE estado IN ('P', 'E');
        SELECT COUNT(*) INTO v_total_empleados FROM empleados;

        DBMS_OUTPUT.PUT_LINE('========================================');
        DBMS_OUTPUT.PUT_LINE('ESTADÍSTICAS DE PKG_TAREAS');
        DBMS_OUTPUT.PUT_LINE('========================================');
        DBMS_OUTPUT.PUT_LINE('Total asignaciones realizadas: ' || g_contador_llamadas);
        DBMS_OUTPUT.PUT_LINE('Total tareas activas:         ' || v_total_tareas_activas);
        DBMS_OUTPUT.PUT_LINE('Total empleados registrados:  ' || v_total_empleados);
        DBMS_OUTPUT.PUT_LINE('========================================');
    END mostrar_estadisticas;

END pkg_tareas;
/
```

**Nota:** Si el package ya existe, usar `CREATE OR REPLACE PACKAGE BODY pkg_tareas` para reemplazar el cuerpo.

```sql
SET SERVEROUTPUT ON;

DECLARE
    v_tarea_id  NUMBER;
    v_emp1_id   NUMBER;
    v_emp2_id   NUMBER;
    v_proy_id   NUMBER;
BEGIN
    SELECT id INTO v_proy_id FROM proyectos WHERE nombre = 'App Móvil Corporativa';

    SELECT id INTO v_emp2_id FROM empleados WHERE nombre = 'Laura Díaz';

    -- Crear varias tareas pendientes
    FOR i IN 1..3 LOOP
        INSERT INTO tareas (proyecto_id, nombre, estado)
        VALUES (v_proy_id, 'Tarea ' || i || ' del package', 'P');
    END LOOP;
    COMMIT;

    -- Obtener IDs de las nuevas tareas
    FOR r IN (SELECT id FROM tareas WHERE nombre LIKE 'Tarea % del package' ORDER BY id) LOOP
        BEGIN
            pkg_tareas.asignar(r.id, v_emp1_id);
        EXCEPTION
            WHEN OTHERS THEN
                DBMS_OUTPUT.PUT_LINE('No se pudo asignar tarea ' || r.id || ': ' || SQLERRM);
        END;
    END LOOP;

    -- Asignar una más a otro empleado
    SELECT id INTO v_tarea_id FROM tareas WHERE nombre = 'Implementar API REST';
    pkg_tareas.asignar(v_tarea_id, v_emp2_id);

    -- Mostrar estadísticas
    pkg_tareas.mostrar_estadisticas;

    COMMIT;
END;
/
```

### Pregunta conceptual

> **¿Qué sucede con `g_contador_llamadas` si abrimos dos sesiones separadas? ¿Por qué?**

Cada sesión de base de datos tiene su **propia copia de las variables de package**. Esto se debe a que las variables de package mantienen **estado de sesión**. Si la sesión A incrementa el contador a 5, la sesión B sigue viendo su contador en 0. No hay compartición entre sesiones. Para contadores globales entre sesiones se necesitaría una tabla, un contexto de aplicación o una Sequence.

## Respuesta 5: Sinónimos para paquetes

### Parte A — Sinónimo privado

```sql
-- Crear sinónimo privado
CREATE SYNONYM PKG_TAREAS_SIN FOR pkg_tareas;

-- Verificar en el diccionario
SELECT synonym_name, table_owner, table_name
FROM user_synonyms
WHERE synonym_name = 'PKG_TAREAS_SIN';

-- Usar el sinónimo
DECLARE
    v_tarea_id NUMBER;
    v_emp_id   NUMBER;
    v_costo    NUMBER;
BEGIN
    SELECT id INTO v_tarea_id FROM tareas WHERE estado = 'P' AND ROWNUM = 1;
    SELECT id INTO v_emp_id FROM empleados WHERE ROWNUM = 1;

    PKG_TAREAS_SIN.asignar(v_tarea_id, v_emp_id);
    v_costo := PKG_TAREAS_SIN.calcular_costo(v_tarea_id);

    DBMS_OUTPUT.PUT_LINE('Tarea asignada vía sinónimo. Costo: ' || v_costo);
    COMMIT;
END;
/
```

### Parte B — Sinónimo público

```sql
-- Crear sinónimo público
CREATE PUBLIC SYNONYM PKG_UTILIDADES FOR pkg_utilidades;

-- Consultar sinónimos accesibles
SELECT synonym_name, table_owner, table_name
FROM all_synonyms
WHERE synonym_name = 'PKG_UTILIDADES';

/* ============================================
   DIFERENCIA ENTRE SIN—NIMO PRevisarBLICO Y PRIVADO:

   - Sinónimo PRIVADO:
     * Visible solo para el usuario que lo crea
     * Se almacena en el esquema del usuario
     * No requiere privilegios especiales para crearlo
     * Consultable en USER_SYNONYMS

   - Sinónimo PRevisarBLICO:
     * Visible para TODOS los usuarios de la BD
     * Se almacena en el esquema PUBLIC
     * Requiere privilegio CREATE PUBLIC SYNONYM
     * Consultable en ALL_SYNONYMS (para todos) o
       DBA_SYNONYMS (para DBA)
     * Revisartil para exponer paquetes corporativos a
       múltiples esquemas sin dar acceso directo
   ============================================ */

-- Demostrar uso del sinónimo público
BEGIN
    DBMS_OUTPUT.PUT_LINE(pkg_utilidades.formatear_moneda(9999.99, 'EUR'));
    DBMS_OUTPUT.PUT_LINE(pkg_utilidades.estado_texto('C'));
END;
/
```

### Parte C — Eliminación

```sql
-- Eliminar sinónimo privado
```
