## Respuesta 2: Trigger AFTER UPDATE — Actualización de estado de proyecto

```sql
/*
 * Justificación: Este trigger debe ser STATEMENT-LEVEL (sin FOR EACH ROW).
 *
 * Motivo: Necesitamos evaluar TODAS las tareas del proyecto después de la
 * operación completa. Si usáramos FOR EACH ROW, evaluaríamos cada fila
 * individualmente y podríamos actualizar el proyecto prematuramente
 * (cuando aún faltan tareas por procesar en el mismo UPDATE).
 *
 * Además, si el UPDATE afecta múltiples proyectos, con statement-level
 * podemos evaluar todos los proyectos afectados en una sola pasada,
 * lo que es más eficiente.
 */

CREATE OR REPLACE TRIGGER trg_actualizar_estado_proyecto
    AFTER UPDATE OF estado ON tareas
DECLARE
    CURSOR c_proyectos IS
        SELECT DISTINCT proyecto_id
        FROM tareas
        WHERE proyecto_id IN (
            SELECT DISTINCT proyecto_id
            FROM tareas
            WHERE estado != 'C'
        )
        MINUS
        SELECT DISTINCT proyecto_id
        FROM tareas
        WHERE estado != 'C';
BEGIN
    FOR r IN c_proyectos LOOP
        UPDATE proyectos
        SET estado = 'C'
        WHERE id = r.proyecto_id
          AND estado != 'C';

        DBMS_OUTPUT.PUT_LINE('Proyecto ' || r.proyecto_id || ' marcado como Completado');
    END LOOP;
END;
/
```

### Prueba

```sql
SET SERVEROUTPUT ON;

DECLARE
    v_proy_id NUMBER;
    v_tarea1  NUMBER;
    v_tarea2  NUMBER;
BEGIN
    SELECT id INTO v_proy_id FROM proyectos WHERE nombre = 'App Móvil Corporativa';

    -- Crear tareas para este proyecto
    INSERT INTO tareas (proyecto_id, nombre, estado)
    VALUES (v_proy_id, 'Tarea A', 'P')
    RETURNING id INTO v_tarea1;

    INSERT INTO tareas (proyecto_id, nombre, estado)
    VALUES (v_proy_id, 'Tarea B', 'P')
    RETURNING id INTO v_tarea2;
    COMMIT;

    -- Completar todas las tareas del proyecto
    UPDATE tareas SET estado = 'C'
    WHERE proyecto_id = v_proy_id AND id IN (v_tarea1, v_tarea2);
    COMMIT;

    -- Verificar que el proyecto cambió a 'C'
    FOR r IN (SELECT nombre, estado FROM proyectos WHERE id = v_proy_id) LOOP
        DBMS_OUTPUT.PUT_LINE('Proyecto: ' || r.nombre || ' | Estado: ' || r.estado);
    END LOOP;

    -- Agregar nueva tarea pendiente
    INSERT INTO tareas (proyecto_id, nombre, estado)
    VALUES (v_proy_id, 'Tarea C (nueva)', 'P');
    COMMIT;

    -- Verificar que el proyecto SIGUE en 'C' (porque la nueva tarea no provocó un UPDATE)
    FOR r IN (SELECT nombre, estado FROM proyectos WHERE id = v_proy_id) LOOP
        DBMS_OUTPUT.PUT_LINE('Proyecto: ' || r.nombre || ' | Estado: ' || r.estado);
    END LOOP;

    -- Si actualizamos la nueva tarea a 'C', el trigger se vuelve a disparar
    UPDATE tareas SET estado = 'C' WHERE nombre = 'Tarea C (nueva)';
    COMMIT;
END;
/
```

## Respuesta 3: Trigger de Auditoría

```sql
CREATE TABLE auditoria_tareas (
    id               NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    tarea_id         NUMBER(10),
    campo_modificado VARCHAR2(50),
    valor_anterior   VARCHAR2(500),
    valor_nuevo      VARCHAR2(500),
    usuario          VARCHAR2(100),
    fecha            TIMESTAMP DEFAULT SYSTIMESTAMP
);

COMMENT ON TABLE auditoria_tareas IS 'Auditoría de cambios en la tabla TAREAS';
COMMENT ON COLUMN auditoria_tareas.id IS 'Identificador único del registro de auditoría';
COMMENT ON COLUMN auditoria_tareas.tarea_id IS 'ID de la tarea modificada';
COMMENT ON COLUMN auditoria_tareas.campo_modificado IS 'Nombre del campo modificado';
COMMENT ON COLUMN auditoria_tareas.valor_anterior IS 'Valor anterior del campo';
COMMENT ON COLUMN auditoria_tareas.valor_nuevo IS 'Valor nuevo del campo';
COMMENT ON COLUMN auditoria_tareas.usuario IS 'Usuario que realizó la modificación';
COMMENT ON COLUMN auditoria_tareas.fecha IS 'Fecha y hora de la modificación';

CREATE OR REPLACE TRIGGER trg_auditar_tareas
    AFTER UPDATE ON tareas
    FOR EACH ROW
BEGIN
    -- Auditar cambio en horas_estimadas
    IF :OLD.horas_estimadas != :NEW.horas_estimadas THEN
        INSERT INTO auditoria_tareas (tarea_id, campo_modificado, valor_anterior, valor_nuevo, usuario)
        VALUES (:OLD.id, 'HORAS_ESTIMADAS',
                TO_CHAR(:OLD.horas_estimadas), TO_CHAR(:NEW.horas_estimadas),
                USER);
    END IF;

    -- Auditar cambio en estado
    IF :OLD.estado != :NEW.estado THEN
        INSERT INTO auditoria_tareas (tarea_id, campo_modificado, valor_anterior, valor_nuevo, usuario)
        VALUES (:OLD.id, 'ESTADO',
                :OLD.estado, :NEW.estado,
                USER);
    END IF;
END;
/
```

```sql
SET SERVEROUTPUT ON;

DECLARE
    v_tarea_id NUMBER;
BEGIN
    SELECT id INTO v_tarea_id FROM tareas WHERE nombre = 'Tarea prueba trigger';

    -- 1. Actualizar solo horas_estimadas
    UPDATE tareas SET horas_estimadas = 150 WHERE id = v_tarea_id;
    COMMIT;

    -- 2. Actualizar solo estado
    UPDATE tareas SET estado = 'C' WHERE id = v_tarea_id;
    COMMIT;

    -- 3. Actualizar ambos campos en un solo UPDATE
    SELECT id INTO v_tarea_id FROM tareas WHERE nombre = 'Tarea A';
    UPDATE tareas SET horas_estimadas = 200, estado = 'E' WHERE id = v_tarea_id;
    COMMIT;

    -- 4. Consultar auditoría
    SELECT a.id, a.tarea_id, a.campo_modificado,
           a.valor_anterior, a.valor_nuevo,
           a.usuario, a.fecha
    FROM auditoria_tareas a
    ORDER BY a.id;
END;
/
```

## Respuesta 4: Trigger Compuesto (12c)

```sql
-- Agregar columna si no existe
ALTER TABLE proyectos ADD horas_totales NUMBER(8,1) DEFAULT 0;

COMMENT ON COLUMN proyectos.horas_totales IS 'Total de horas reales acumuladas de todas las tareas del proyecto';

CREATE OR REPLACE TRIGGER trg_control_tiempo
    FOR UPDATE OF horas_reales ON tareas
    COMPOUND TRIGGER

    -- Variable global del trigger (persiste entre secciones)
    TYPE t_acumulador IS TABLE OF NUMBER(8,1) INDEX BY proyectos.id%TYPE;
    v_acumulador t_acumulador;

    BEFORE EACH ROW IS
    BEGIN
        -- Validar que horas_reales no exceda el límite
        IF :NEW.horas_reales IS NOT NULL AND :NEW.horas_estimadas > 0 THEN
            IF :NEW.horas_reales > :NEW.horas_estimadas * 1.5 THEN
                RAISE_APPLICATION_ERROR(-20100,
                    'Las horas reales (' || :NEW.horas_reales ||
                    ') exceden el 150% de las horas estimadas (' ||
                    :NEW.horas_estimadas || ')');
            END IF;
        END IF;
    END BEFORE EACH ROW;

    AFTER EACH ROW IS
    BEGIN
        -- Acumular la diferencia de horas por proyecto
        v_acumulador(:NEW.proyecto_id) := NVL(v_acumulador(:NEW.proyecto_id), 0)
                                          + (:NEW.horas_reales - NVL(:OLD.horas_reales, 0));
    END AFTER EACH ROW;

    AFTER STATEMENT IS
        v_idx proyectos.id%TYPE;
    BEGIN
        -- Actualizar horas_totales en PROYECTOS para cada proyecto afectado
        v_idx := v_acumulador.FIRST;
        WHILE v_idx IS NOT NULL LOOP
            UPDATE proyectos
            SET horas_totales = NVL(horas_totales, 0) + v_acumulador(v_idx)
            WHERE id = v_idx;

            DBMS_OUTPUT.PUT_LINE('Proyecto ' || v_idx ||
                                 ': horas_totales incrementadas en ' || v_acumulador(v_idx));
            v_idx := v_acumulador.NEXT(v_idx);
        END LOOP;
    END AFTER STATEMENT;

END trg_control_tiempo;
/
```

### Prueba

```sql
SET SERVEROUTPUT ON;

DECLARE
    v_proy_id  NUMBER;
    v_tarea1   NUMBER;
    v_tarea2   NUMBER;
    v_horas_actuales NUMBER;
BEGIN
    SELECT id INTO v_proy_id FROM proyectos WHERE nombre = 'Migración a Cloud';

    -- Estado inicial
    SELECT horas_totales INTO v_horas_actuales
    FROM proyectos WHERE id = v_proy_id;
    DBMS_OUTPUT.PUT_LINE('Horas totales iniciales del proyecto: ' || NVL(v_horas_actuales, 0));

    -- Crear tareas
    INSERT INTO tareas (proyecto_id, nombre, horas_estimadas, horas_reales, estado)
    VALUES (v_proy_id, 'Migración BD', 120, 0, 'E')
    RETURNING id INTO v_tarea1;

    INSERT INTO tareas (proyecto_id, nombre, horas_estimadas, horas_reales, estado)
    VALUES (v_proy_id, 'Migración Servidores', 80, 0, 'E')
    RETURNING id INTO v_tarea2;
    COMMIT;

    -- Actualizar horas_reales de ambas tareas en un solo UPDATE
    -- Tarea 1: 50h, Tarea 2: 30h → Total: 80h
    UPDATE tareas
    SET horas_reales = CASE id
                           WHEN v_tarea1 THEN 50
                           WHEN v_tarea2 THEN 30
                       END
    WHERE id IN (v_tarea1, v_tarea2);
    COMMIT;

    -- Verificar horas_totales en proyecto
    SELECT horas_totales INTO v_horas_actuales
    FROM proyectos WHERE id = v_proy_id;
    DBMS_OUTPUT.PUT_LINE('Horas totales después de actualizar: ' || NVL(v_horas_actuales, 0));

    -- Prueba de validación: exceder límite
    BEGIN
        UPDATE tareas SET horas_reales = 200 WHERE id = v_tarea1;
    EXCEPTION
        WHEN OTHERS THEN
            DBMS_OUTPUT.PUT_LINE('Error esperado: ' || SQLERRM);
    END;
    ROLLBACK;
END;
/
```

### Respuesta a pregunta extra

**Ventaja del trigger compuesto frente a triggers separados:**

1. **Una sola unidad lógica**: Todo el código relacionado (validación + acumulación + actualización) está en un solo trigger, facilitando el mantenimiento.

2. **Variable compartida**: La variable `v_acumulador` es privada del trigger compuesto y persiste entre secciones. Con triggers separados necesitaríamos un package con variable de estado o una tabla temporal para pasar datos entre el trigger row-level y el statement-level.

3. **Transaccionalidad**: Como es un solo trigger, Oracle lo maneja como una unidad atómica. Con triggers separados, el orden de ejecución es menos predecible (aunque se puede controlar con `FOLLOWS`).

4. **Rendimiento**: No hay cambio de contexto entre triggers; todo se ejecuta dentro del mismo flujo de código.

5. **Claridad**: La estructura `BEFORE EACH ROW` / `AFTER EACH ROW` / `AFTER STATEMENT` muestra explícitamente el ciclo de vida completo de la operación DML.
